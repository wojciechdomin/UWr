# Usage: python3 interactor.py import sys import argparse parser = argparse.ArgumentParser() parser.add_argument("targetX", type=int, help="X coordinate of the temple") parser.add_argument("targetY", type=int, help="Y coordinate of the temple") args = parser.parse_args() X = 0 Y = 0 V = 1 dr = [1, 0] targetX = args.targetX targetY = args.targetY def printdis(): D = abs(X - targetX) + abs(Y - targetY) print(f"{D}", flush=True) if D == 0: print("[not part of actual interaction] Found temple!") sys.exit(0) while True: x = input() if x == "? L": dr[0], dr[1] = dr[1], dr[0] dr[0] *= -1 elif x == "? R": dr[0] *= -1 dr[0], dr[1] = dr[1], dr[0] elif x == "? F": pass else: print("[not part of actual interaction] Invalid move! Must be '? L', '? R' or '? F'.") sys.exit(1) X += dr[0] * V Y += dr[1] * V printdis() V += 1 if V == 2 * pow(10, 4): print("[not part of actual interaction] Too fast!") sys.exit(1)
